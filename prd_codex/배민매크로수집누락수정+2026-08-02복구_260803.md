def resolve_down_dir() -> Path:
    env_path = os.getenv("DOWN_DIR")

    return fallback

    items: list[dict[str, str]] = []
    seen: set[Path] = set()
    source_dirs = (
        (DOWN_DIR, "down"),
        (DOWN_DIR / "업로드_temp", "down"),
        (COLLECT_SRC, "collect"),
    )
    for source_dir, source_name in source_dirs:
        if not source_dir.exists():
            continue
        for csv_path in sorted(source_dir.glob(f"baemin_{prefix}_*.csv")):
            resolved = csv_path.resolve()
            if resolved in seen:
                continue
            seen.add(resolved)
            items.append({"path": str(csv_path), "source": source_name})
    return items

def _cleanup_loaded_files(loaded_files: list) -> int:
    moved = 0
    for item in loaded_files:
        if isinstance(item, dict):
            path = Path(item.get("path", ""))
            source = item.get("source", "collect")
        else:
            path = Path(str(item))
            source = "collect"
        if not path.exists():
            logger.info("이미 이동됨 또는 삭제됨: %s", path.name)
            continue
        target = _down_collect_path(path) if source == "down" else _archive_path(path)
        try:
            shutil.move(str(path), str(target))
            logger.info("보관됨: %s -> %s", path.name, target)
            moved += 1
        except Exception as exc:
            logger.warning("보관 실패: %s / %s", path.name, exc)
    return moved

def _read_total_summary(driver) -> dict | None:
    """현재 필터 조건의 TotalSummary(건수, 금액)를 읽는다.

        let count = null, amount = null;
        for (const span of spans) {
            const b = span.querySelector('b.TotalSummary-module__jikm');
            if (!b) continue;
            const val = parseInt(b.textContent.replace(/,/g, ''));
            const fullText = span.textContent.trim();
            if (fullText.endsWith('건')) count = val;
            else if (fullText.endsWith('원')) amount = val;
        }
        return (count !== null) ? {count, amount} : null;
        """
    )

def _page_signature(driver) -> str:
    """현재 페이지 식별자: 첫 주문ID|마지막 주문ID|행수."""
    return driver.execute_script(
        """
        const rows = document.querySelectorAll('tr.Table_b_r4ax_1dwbr4on[data-index]');
        if (!rows.length) return '';
        const idOf = (r) => r.querySelector('td[data-td-index="1"]')?.textContent.trim() || '';
        return idOf(rows[0]) + '|' + idOf(rows[rows.length - 1]) + '|' + rows.length;
        """

def _wait_for_page_transition(driver, prev_sig: str) -> bool:
    """페이지 시그니처(_page_signature)가 바뀌면 전환 완료로 본다."""
    start = time.time()
    while time.time() - start < _PAGE_TRANSITION_TIMEOUT:
